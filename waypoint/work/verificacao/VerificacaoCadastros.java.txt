package main;

import Entities.*;
import java.math.BigDecimal;
import java.time.LocalDateTime;

public class VerificacaoCadastros {
    private static int verificacoes;
    public static void main(String[] args) {
        conferir(TesteSistema.validarTexto("  Calcada  ", 20).equals("Calcada"));
        rejeitar(() -> TesteSistema.validarTexto("   ", 20));
        rejeitar(() -> TesteSistema.validarTexto("x".repeat(21), 20));
        conferir(TesteSistema.validarDecimal("-23,5505200", "-90", "90", 7)
                .compareTo(new BigDecimal("-23.55052")) == 0);
        conferir(TesteSistema.validarDecimal("99.99", "0", "99.99", 2)
                .compareTo(new BigDecimal("99.99")) == 0);
        conferir(TesteSistema.validarDecimal("180", "-180", "180", 7).intValueExact() == 180);
        rejeitar(() -> TesteSistema.validarDecimal("90.0000001", "-90", "90", 7));
        rejeitar(() -> TesteSistema.validarDecimal("-180.1", "-180", "180", 7));
        rejeitar(() -> TesteSistema.validarDecimal("100", "0", "99.99", 2));
        rejeitar(() -> TesteSistema.validarDecimal("-1", "0", "99.99", 2));
        rejeitar(() -> TesteSistema.validarDecimal("1.234", "0", "99.99", 2));
        rejeitar(() -> TesteSistema.validarDecimal("NaN", "0", "99.99", 2));
        rejeitar(() -> TesteSistema.validarDecimal("1e10", "0", "99.99", 2));
        rejeitar(() -> TesteSistema.validarDecimal("1,2.3", "0", "99.99", 2));
        cancelar(() -> TesteSistema.validarTexto(null, 20));
        cancelar(() -> TesteSistema.validarDecimal(null, "0", "99.99", 2));
        Usuario u = new Usuario(1L, "Teste", "SEGREDO_TESTE", "teste@exemplo.com", "s");
        Categoria c = new Categoria(1L, "Calcada", "Acessibilidade");
        Trecho t = new Trecho(1L, "Rua de teste", new BigDecimal("-23.55"), new BigDecimal("-46.63"), "Centro");
        Relato r = new Relato(1L, LocalDateTime.of(2026, 11, 17, 14, 30), "Buraco", "aberto", u, t, c);
        Rota rota = new Rota(1L, new BigDecimal("1.25"), u, t, t);
        Vldc_Relato voto = new Vldc_Relato(1L, "s", u, r);
        conferir(r.getUsuario() == u && r.getTrecho() == t && r.getCategoria() == c);
        conferir(rota.getTrechoInicio() == t && rota.getTrechoFinal() == t);
        conferir(voto.getRelato() == r && voto.getUsuario() == u);
        for (Object objeto : new Object[]{u, c, t, r, rota, voto}) {
            conferir(!objeto.toString().contains("SEGREDO_TESTE"));
        }
        System.out.println("OK: " + verificacoes + " verificacoes de limites, cancelamento e relacionamentos.");
    }
    private static void conferir(boolean resultado) {
        if (!resultado) throw new AssertionError("Falha na verificacao " + verificacoes);
        verificacoes++;
    }
    private static void rejeitar(Runnable acao) {
        try { acao.run(); } catch (IllegalArgumentException e) { verificacoes++; return; }
        throw new AssertionError("Entrada invalida aceita");
    }
    private static void cancelar(Runnable acao) {
        try { acao.run(); } catch (RuntimeException e) {
            conferir(e.getClass().getSimpleName().equals("CadastroCancelado")); return;
        }
        throw new AssertionError("Cancelamento nao tratado");
    }
}
